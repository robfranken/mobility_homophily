general custom functions

  • fpackage.check: Check if packages are installed (and install if not) in R
  • fsave: Function to save data with time stamp in correct directory
  • fload: Function to load R-objects under new names
  • fshowdf: Print objects (tibble / data.frame) nicely on screen in .Rmd.
  • ffit: fit a series of (here, generalized linear mixed-effects) models
fpackage.check <- function(packages) {
    lapply(packages, FUN = function(x) {
        if (!require(x, character.only = TRUE)) {
            install.packages(x, dependencies = TRUE)
            library(x, character.only = TRUE)
        }
    })
}

fsave <- function(x, file, location = "./data/processed/", ...) {
    if (!dir.exists(location))
        dir.create(location)
    datename <- substr(gsub("[:-]", "", Sys.time()), 1, 8)
    totalname <- paste(location, datename, file, sep = "")
    print(paste("SAVED: ", totalname, sep = ""))
    save(x, file = totalname)
}

fload <- function(fileName) {
    load(fileName)
    get(ls()[ls() != "fileName"])
}

fshowdf <- function(x, digits = 2, ...) {
    knitr::kable(x, digits = digits, "html", ...) %>%
        kableExtra::kable_styling(bootstrap_options = c("striped", "hover")) %>%
        kableExtra::scroll_box(width = "100%", height = "300px")
}

ffit <- function(formula, data) {
    tryCatch({
        model <- lme4::glmer(formula, data = data, family = binomial(link = "logit"), control = glmerControl(optimizer = "bobyqa",
            optCtrl = list(maxfun = 1e+05)))
        cat("Fitting model:", as.character(formula), "\n")
        summary(model)
        cat("\n")
        return(model)
    }, error = function(e) {
        cat("Error fitting model:", as.character(formula), "\n")
        cat("Error message:", conditionMessage(e), "\n")
        return(NULL)
    })
}

who maintains educationally (dis)similar friends and confidants, and does this depend on one’s mobility trajectory?

# function to calculate proportion self-similar and dis-similar alters that is maintained (and
# their ratio), for a given group (e.g., uni), tie type (e.g, friend), period (e.g., first period),
# cohort (e.g., first years), and mobility trajectory (e.g., upwardly mobile)

fprops <- function(df, ego_educ = NULL, tie = NULL, mobility = NULL, period = NULL, year = NULL) {
    subset_df <- df
    # apply filters if specified
    if (!is.null(ego_educ)) {
        subset_df <- subset_df[subset_df$ego_educ == ego_educ, ]
    }
    if (!is.null(tie)) {
        subset_df <- subset_df[subset_df$tie == tie, ]
    }
    if (!is.null(mobility)) {
        subset_df <- subset_df[subset_df$mobility == mobility, ]
    }
    if (!is.null(period)) {
        subset_df <- subset_df[subset_df$period == period, ]
    }
    if (!is.null(year)) {
        subset_df <- subset_df[subset_df$study.year == year, ]
    }

    # means for sim_educ == 1 and sim_educ == 0 and the ratio
    mean_sim1 <- mean(subset_df[subset_df$sim_educ == 1, ]$Y, na.rm = TRUE)
    n1 <- nrow(subset_df[subset_df$sim_educ == 1, ])
    mean_sim0 <- mean(subset_df[subset_df$sim_educ == 0, ]$Y, na.rm = TRUE)
    n0 <- nrow(subset_df[subset_df$sim_educ == 0, ])
    mean_ratio <- mean_sim1/mean_sim0

    return(list(mean_sim1 = mean_sim1, mean_sim0 = mean_sim0, mean_ratio = mean_ratio, n1 = n1, n0 = n0))
}
